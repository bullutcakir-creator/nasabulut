app.post("/ai/image" , async (req, res) => {
    const { prompt, size } = req.body;
    if (typeof prompt !== 'string' || !prompt.trim()) {
        return res.status(400).json({ error: 'Prompt is required' });
    }
    if (size && !['256x256', '512x512', '1024x1024'].includes(size)) {
        return res.status(400).json({ error: 'Size must be one of 256x256, 512x512, or 1024x1024' });
    }

    try {
        const aiResponse = await fetch(`${aiBackendUrl.replace(/\/+$/, '')}/v1/images/generations`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                prompt: prompt.trim(),
                size: size || '512x512'
            }),
            signal: AbortSignal.timeout(aiTimeoutMs)
        });
        const responseBody = await aiResponse.text();
        let responseData;

        try {
            responseData = JSON.parse(responseBody);
        } catch {
            throw Object.assign(new Error('AI backend returned an invalid JSON response'), { status: 502 });
        }

        if (!aiResponse.ok) {
            throw Object.assign(new Error('AI backend request failed'), { status: 502, details: responseData });
        }

        return res.json(responseData);
    } catch (error) {
        return sendAiError(res, error);
    }
});
